\begin{frame}{Tail risk}
    \begin{wideitemize}
    \item Tail events \textcolor{blue}{rarely happen} \textcolor{red}{but do matter.}
    \item Rietz (1988) and Barro (2006) show that rare events significantly contribute to the equity premium.
    \item Disastrous events also shape economic agents' lifetime expectations and behaviors (Bernile, Bhagwat, Rau, 2016; Gao, Liu, Shi, 2020).
    \begin{itemize}
    \item \textcolor{blue}{This implies that agents may have distinct beliefs about the probability of rare events.}
    \end{itemize}
    \item Measuring tail risk and market participants' perceptions of tail risk is a crucial question in financial economics.
    \end{wideitemize}
\end{frame}

\begin{frame}{Literature on tail risk}
    \begin{wideitemize}
    \item There is a substantial body of work on how to capture tail risk.
    \bigskip
    \begin{itemize}
        \item \textcolor{blue}{Options}: Backus, Chernov, and Martin (2011) employ options to infer disastrous events. The most famous example is the VIX index.
    \bigskip
        \item \textcolor{blue}{Cross-sectional returns}: Kelly and Jiang (2014) use monthly firm-level price crashes to identify common fluctuations in tail risk among individual stocks.
    \bigskip
        \item \textcolor{blue}{News}: Manela and Moreira (2017) employ a text-based method to measure news-implied tail risk (NVIX).
    \end{itemize}
    \end{wideitemize}
\end{frame}

\begin{frame}{How do different participants perceive the tail events?}
    \bigskip
    \begin{wideitemize}
    \item However, \alert{we still know little about how different participants perceive tail risk.}
    \bigskip
    \begin{enumerate}
    \item \textcolor{blue}{Rare} events are, by definition, \textcolor{green}{challenging to estimate their data-generating process} (Barro, 2006).
    %\bigskip
    %\item It's unrealistic to survey every participant about their beliefs on tail events through questionnaires.
    % \item Nearly all works on disagreement focus on individual market agents' first-order estimates (averages) of economic variables. They compare their estimates to reach a consensus, such as EPS, GDP, CPI, etc.
    \bigskip  
    \item We seldom have direct measures of agents' beliefs about the entire distribution, including \alert{tail events}. 
    \end{enumerate}
    \end{wideitemize}
\end{frame}

        \item How can we \alert{capture} disagreement on tail (DOT) in the equity market?
        \bigskip
        \item How does DOT relate to \alert{asset pricing} (and its economic mechanism)?

    \end{wideitemize}
\end{frame}

\begin{frame}{Methods: a Road Map}
    \bigskip
    \begin{wideitemize}
    \item Heterogeneous beliefs about tail events are generated by \textcolor{blue}{various machine agents} (Bali et al., 2023) .
    
    \item To capture different investors' beliefs about tail events, we propose a statistical \alert{classification} method (He, Lyu and Zhou, 2024).

    \item We use the \textcolor{blue}{standard deviation} of forecasts on tail events by different investors (Banerjee 2011; Yu, 2011).
    \end{wideitemize}
\end{frame}

\begin{frame}{Methods: Source of Disagreement }
    \begin{quote}
    If two people are to disagree, one of the following must be true.
    \bigskip
    \begin{enumerate}
        \item \alert{They have different priors beliefs.}
    \bigskip
        \item They observe different information.
    \bigskip
        \item one or both of them is not full rational.
    \end{enumerate}
    \bigskip
    - Barberis (2018).
    \end{quote}
\end{frame}


\begin{frame}{Methods: Differentiated Investors}
    \bigskip
    \begin{wideitemize}
    \item To model the belief of individual investor $n$, we assume all of the investors observe a \alert{public, complete, and identical} information set $Z = (z_{i, t})$, where $z_{i,t}=(z_{i,t}^{1},z_{i,t}^{2},\cdots,z_{i,t}^{K})'$ is a 94-dimensional vector representing 94 firm characteristics of stock $i$ in month $t$. 
    % \bigskip
    \item The divergence in beliefs $X_{n} = (x_{n,i,t})$ is generated by assigning different weights to each characteristic:
    \begin{equation*}
        x_{n,i,t} = g_n(z_{i,t}) = D_{n}z_{i,t}
    \end{equation*}
    where $D_{n}$ is a diagonal matrix of rank 94, with each diagonal element following:
    \begin{equation*}
        d_{j,j} \sim N(0,1), \text{I.I.D.}
    \end{equation*}
    \end{wideitemize}
\end{frame}

\begin{frame}{Methods: Learning Target}
    \bigskip
    \begin{wideitemize}
    \item In Gu, Kelly, and Xiu (2020), they consider a general conditional risk premium formulation:
    \begin{equation*}
        E_t[R_{i,t+1}] = f(z_{i,t})
    \end{equation*}
    \item To account for disagreement on extreme events, we modify their methods. \alert{Instead of focusing on regression, we approach it as a classification task}. We define four events: Tail (T), Down (D), Up (U), and Surge (S), corresponding to monthly expected returns in ($-\infty$, -25\%), [-25\%, 0\%), [0\%, 25\%), [25\%, $+\infty$), respectively.
    \item We frame the general conditional risk premium in a \alert{classification format}:
    \begin{equation*}
        E_t[P_{i,t+1}] = f(z_{i,t})
    \end{equation*}
    where $P_{i,t+1} =(P_{1},P_{2},P_{3},P_{4})'$ are the probabilities for the 4 events.
    \end{wideitemize}
\end{frame}

\begin{frame}{Methods: Neural Network Model}
    \bigskip
    \begin{wideitemize}
    \item In this study, to address the complexity of the investor's belief formation process, we employ the \textcolor{blue}{Neural Network Model (NNM)} to express $f(x_{n,i,t})$.
    \item NNM is a computational model inspired by the way \alert{biological neural networks} in the human brain process information. It is often considered the most powerful tool in machine learning and is theoretically grounded as \textcolor{blue}{``universal approximator"} for any smooth predictive relationship (Hornik et al, 1989; Cybenko, 1989)
    % \item The NNM is a four-class classification feedforward neural network designed to classify input data into one of four classes and provide probabilities for each class. It comprises input, hidden, and output layers, and learns by adjusting weights and biases through backpropagation. It utilizes activation functions and optimization algorithms during training. Once trained, it can predict class labels for new data.
    \item Our NNM is a \alert{four-class classification feedforward neural network model}, with the model setup similar to that of Gu, Kelly, and Xiu (2020). 
    \end{wideitemize}
\end{frame}

\begin{frame}{Methods: Neural Network Model}
    \bigskip
    \begin{wideitemize}
    \item Consider a NNM with one input layer, $L$ hidden layer, and one output layer. Each hidden layer uses the \alert{ReLU} (Rectified Linear Unit) activation function\footnote{The ReLU activation function is defined as: $ReLU(u^{(l)}_i)=\max(0,u^{(l)}_i)$.}, and the output layer uses the \alert{softmax} activation function\footnote{The softmax activation function is defined as: $softmax(u^{(L+1)}_i)=\frac{e^{u^{(L+1)}_i}}{\sum_{j=1}^Me^{u^{(L+1)}_j}}$, where $u^{(L+1)}_i$ is the $i$-th element of the unactivated input $\mathbf{u}^{(L+1)}$, and $M$ is the total number of classes.}, which is typically used for multi-class classification problems.
    \begin{equation*}
E_{n,t}[P_{i,t+1}]=NN_{n}\left(x_{n,i,t}\right)=softmax\left(W^{(L+1)}a^{(L)}+b^{(L+1)}\right) \label{nn}
    \end{equation*}
    where:$$a^{(l)}=ReLU(u^{(l)}),\quad u^{(l)}=W^{(l)}a^{(l-1)}+b^{(l)},\quad a^{(0)}={x_{n,i,t}},\quad l=1,2,\ldots,L$$
    \end{wideitemize}
\end{frame}

\begin{frame}{Methods: DOT}
    \bigskip
    \begin{wideitemize}
    \item For stock $i$ and month $t$, after generating beliefs on T, D, U and S from 100 machine investors, we employ standard deviation to capture the \alert{disagreement on tail (DOT)}:
    \begin{equation*}
        DOT_{i,t}=\sqrt{\sum_{n=1}^{N}\frac{\left(E_{n,t}[P_{i,t+1,1}]-E_t[P_{i,t+1,1}]\right)^{2}}{N-1}}.
    \end{equation*}
    where $E_t[P_{i,t+1,1}] = \sum_{n=1}^{N}\frac{E_{n,t}[P_{i,t+1,1}]}{N}$ represents the mean of machine investors' belief of probability on tail.
    \end{wideitemize}
\end{frame}

\begin{frame}{Conclusions}
\begin{wideitemize}
    \item We propose a statistical \textcolor{red}{classification} method to capture investors' heterogeneous beliefs about tail events (DOT).
    \item DOT \textcolor{blue}{negatively} predicts future stock returns, with this negative relationship primarily driven by the \textcolor{orange}{mispricing} channel (Miller, 1977).
    \item We validate the effectiveness of DOT through the \textcolor{cyan}{risk-sharing} framework proposed by Chen et al. (2012).
    \item The anatomy of disagreement reveals that DOT plays a \textcolor{green}{crucial} role, outperforming disagreement on other events.
\end{wideitemize}
\end{frame}
