\begin{frame}{Belief on Tail Events}
    \begin{wideitemize}
    \item Tail events \textcolor{blue}{rarely happen} but \textcolor{red}{do matter.}
    \item Rietz (1988) and Barro (2006) demonstrate that rare events significantly contribute to the equity premium.
    \item Disastrous events also shape economic agents' lifetime expectations and behaviors (Malmendier and Nagel, 2011; Bernile, Bhagwat, Rau, 2016).
    \begin{itemize}
    \item \textcolor{blue}{This implies that agents have \textcolor{red}{dispersed beliefs} on the probability of rare events.}
    \end{itemize}
  
    %\item Measuring tail risk and market participants' perceptions of tail risk is a crucial question in financial economics.
    %\item Even without an actual tail event, expectations about rare events strongly influence asset pricing (Liu, Pan, Wang, 2005)
    
    \end{wideitemize}
\end{frame}

\begin{frame}{Disagreement}
    \begin{wideitemize}
    \item Investor disagreement (i.e., heterogeneous beliefs) affects asset prices, volatility, and trading volume
      (e.g., Miller 1977;  Duffie and Heaton, 1996; Hong and Sraer 2016; Atmaz and Basak 2018).
   
    %\item Tail events are hard to assess, leading to heterogeneous beliefs among investors. \textbf{Disagreement on tail event} can cause sharp valuation and risk-sharing shifts (Chen, Joslin, Tran 2012).
     \item Theoretically, \textcolor{red}{\textbf{disagreement on tail events}} has important implications for stock valuation and risk-sharing (Chen, Joslin, Tran 2012).
    
    % \item Understanding differences in perceived tail risk is crucial for explaining 
    % \begin{wideitemize}
    %     \item Return dynamics during crises and calm periods. 
    %     \item Fluctuation of perceived tail risk.
    % \end{wideitemize}

    \item But, how to \textcolor{blue}{empirically measure} disagreement on tail events?
  %  \item Calling for a measure to capture the \textbf{disagreement on tail events}.
    \end{wideitemize}
\end{frame}

\begin{frame}{Existing Measures on Tail Risk and Disagreement}
    \begin{wideitemize}
    \item There is a large body of work on how to capture tail risk.
    \begin{wideitemize}
        \item \textcolor{blue}{Options}: The most famous example is the VIX. Backus, Chernov, and Martin (2011) employ options to infer disastrous events. 
        \vspace{-0.2 cm}
        \item \textcolor{blue}{Cross-sectional returns}: Kelly and Jiang (2014) use monthly firm-level price crashes to identify common fluctuations in tail risk among individual stocks.
         \vspace{-0.2 cm}
        \item \textcolor{blue}{News}: Manela and Moreira (2017) use textual analysis to measure news-implied tail risk.
    \end{wideitemize}
    \item There is also a substantial body of work on how to capture disagreement.
    \begin{wideitemize}
        \item \textcolor{blue}{Forecast dispersion}: Diether, Malloy, Scherbina (2002) use analysts' earning forecast dispersion
         \vspace{-0.2 cm}
        \item \textcolor{blue}{Trading volume}: Garfinkel (2009) uses unexplained trading volume
         \vspace{-0.2 cm}
        \item \textcolor{blue}{Option open interest divergence}: Ge, Lin, Pearson (2016) use option put-call open interest divergence
    \end{wideitemize}
    \end{wideitemize}
\end{frame}

\begin{frame}{Methods: Differentiated Investors}
    \bigskip
    \begin{wideitemize}
    \item To model the belief of individual investor $n$, we assume all of the investors observe a \alert{public, complete, and identical} information set $Z = (z_{i, t})$, where $z_{i,t}=(z_{i,t}^{1},z_{i,t}^{2},\cdots,z_{i,t}^{K})'$ is a 94-dimensional vector representing 94 firm characteristics of stock $i$ in month $t$. 
    % \bigskip
    \item The divergence in beliefs $X_{n} = (x_{n,i,t})$ is generated by assigning different weights to each characteristic:
    \begin{equation*}
        x_{n,i,t} = g_n(z_{i,t}) = D_{n}z_{i,t}
    \end{equation*}
    where $D_{n}$ is a diagonal matrix of rank 94, with each diagonal element following:
    \begin{equation*}
        d_{j,j} \sim N(0,1), \text{I.I.D.}
    \end{equation*}
    \end{wideitemize}
\end{frame}

\begin{frame}{Methods: Learning Target}
    \bigskip
    \begin{wideitemize}
    \item In Gu, Kelly, and Xiu (2020), they consider a general conditional risk premium formulation:
    \begin{equation*}
        E_t[R_{i,t+1}] = f(z_{i,t})
    \end{equation*}
    \item To account for disagreement on extreme events, we modify their methods. \alert{Instead of focusing on regression, we approach it as a classification task}. We define four events: Tail (T), Down (D), Up (U), and Surge (S), corresponding to monthly expected returns in ($-\infty$, -25\%), [-25\%, 0\%), [0\%, 25\%), [25\%, $+\infty$), respectively.
    \item We frame the general conditional risk premium in a \alert{classification format}:
    \begin{equation*}
        E_{n,t}[\mathbf{1}_{i,t+1}] = f_{n,t}(x_{n,i,t})
    \end{equation*}
    where $\mathbf{1}_{i,t+1}=(1_{R_{i,t+1}\in T},
                  1_{R_{i,t+1}\in D},
                  1_{R_{i,t+1}\in U},
                  1_{R_{i,t+1}\in S})'$ is the indicator vector, whose expectation gives the probabilities for the 4 events.
    \end{wideitemize}
\end{frame}

\begin{frame}{Methods: Neural Network Model}
    \bigskip
    \begin{wideitemize}
    \item In this study, to accommodate the complexity of investors' belief formation process, we employ the \textcolor{blue}{Neural Network Model (NNM)} to express $f_{n,t}(x_{n,i,t})$.
    \item NNM is a computational model inspired by the way \alert{biological neural networks} process information in the human brain. It is often considered the most powerful tool in machine learning and is theoretically grounded as \textcolor{blue}{``universal approximator"} for any smooth predictive relationship (Hornik et al., 1989; Cybenko, 1989)
    % \item The NNM is a four-class classification feedforward neural network designed to classify input data into one of four classes and provide probabilities for each class. It comprises input, hidden, and output layers, and learns by adjusting weights and biases through backpropagation. It utilizes activation functions and optimization algorithms during training. Once trained, it can predict class labels for new data.
    \item Our NNM is a \alert{four-class classification feedforward neural network model}, with the model setup similar to Gu, Kelly, and Xiu (2020). 
    \end{wideitemize}
\end{frame}

\begin{frame}{Methods: Neural Network Model}
    \bigskip
    \begin{wideitemize}
    \item Consider a NNM with one input layer, $L$ hidden layer, and one output layer. Each hidden layer uses the \alert{ReLU} (Rectified Linear Unit) activation function\footnote{The ReLU activation function is defined as: $ReLU(u^{(l)}_i)=\max(0,u^{(l)}_i)$.}, and the output layer uses the \alert{softmax} activation function\footnote{The softmax activation function is defined as: $softmax(u^{(L+1)}_i)=\frac{e^{u^{(L+1)}_i}}{\sum_{j=1}^Me^{u^{(L+1)}_j}}$, where $u^{(L+1)}_i$ is the $i$-th element of the unactivated input $\mathbf{u}^{(L+1)}$, and $M$ is the total number of classes.}, which is typically used for multi-class classification problems.
    \begin{equation*}
E_{n,t}[\mathbf{1}_{i,t+1}]=NN_{n,t}\left(x_{n,i,t}\right)=softmax\left(W^{(L+1)}a^{(L)}+b^{(L+1)}\right) \label{nn}
    \end{equation*}
    where:$$a^{(l)}=ReLU(u^{(l)}),\quad u^{(l)}=W^{(l)}a^{(l-1)}+b^{(l)},\quad a^{(0)}={x_{n,i,t}},\quad l=1,2,\ldots,L$$
    \end{wideitemize}
\end{frame}

\begin{frame}{Methods: DOT}
    \bigskip
    \begin{wideitemize}
    \item For stock $i$ and month $t$, after generating beliefs on T, D, U and S from 100 machine investors, we employ standard deviation to capture the \alert{disagreement on tail (DOT)}:
    \begin{equation*}
        DOT_{i,t}=\sqrt{\frac{1}{N-1}\sum_{n=1}^{N}\left(E_{n,t}[1_{R_{i,t+1}\in T}]-E_t[1_{R_{i,t+1}\in T}]\right)^{2}}
    \end{equation*}
       
    where $E_t[1_{R_{i,t+1}\in T}] = \frac{1}{N}\sum_{n=1}^{N}E_{n,t}[1_{R_{i,t+1}\in T}]$ represents the mean of machine investors' beliefs on the tail event probability.
   
    %where $E_t[P_{i,t+1,1}] = \sum_{n=1}^{N}\frac{E_{n,t}[P_{i,t+1,1}]}{N}$ represents the mean of machine investors' belief of probability on tail.
    \end{wideitemize}
\end{frame}

\begin{frame}{Conclusions}
\begin{wideitemize}
    \item We propose a statistical \textcolor{red}{classification} method to capture investors' heterogeneous beliefs about tail events (DOT).
    \item DOT \textcolor{blue}{negatively} predicts future stock returns, consistent with a \textcolor{purple}{mispricing} channel (Miller, 1977).
    \item The effectiveness of DOT is validated through the \textcolor{cyan}{risk-sharing} framework (Chen et al., 2012).
    \item The anatomy of disagreement reveals that DOT plays a \textcolor{green}{crucial} role, outweighing disagreement on other events.
\end{wideitemize}
\end{frame}
