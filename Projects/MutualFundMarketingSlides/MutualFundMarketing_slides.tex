\section{Motivation}
% ===========================================================================

\begin{frame}{Mutual Fund Marketing: What We Know}

\textbf{Asset management has grown dramatically:}
\begin{itemize}
  \item U.S.\ AUM and fund count have expanded sharply over three decades
  \item Investor search costs are a first-order friction in capital allocation \citep{hortacsu2004,huang2007participation}
\end{itemize}

\vspace{0.4cm}
\textbf{Marketing successfully attracts fund-level inflows:}
\begin{itemize}
  \item Funds spend roughly \hl{one-third of fee revenue} on marketing
  \item Advertising, broker distribution, and visibility drive significant inflows
  \citep{sirri1998costly,gallaher2008advertising,huang2007participation}
\end{itemize}

\vspace{0.4cm}
\begin{keybox}
\textbf{Implicit assumption in the literature:} fund-level inflows $\Rightarrow$ family-level growth.
\end{keybox}

\end{frame}

% ---------------------------------------------------------------------------

\begin{frame}{The Gap: Fund vs.\ Family Level}

\textbf{Is the implicit assumption correct?}

\medskip
If marketing \emph{reallocates} capital within families rather than attracting new investors:
\begin{itemize}
  \item Fund-level effects do \emph{not} translate into family growth
  \item Standard fund-level regressions overstate marketing's role
  \item Families' incentives to market may be entirely different from what we think
\end{itemize}

\vspace{0.4cm}
\textbf{Testing this requires:}
\begin{enumerate}
  \item A novel marketing channel with \hl{sharp within-family variation}
  \item Joint observation of fund-level \emph{and} family-level flows
  \item A setting where we can cleanly identify spillovers to non-promoted funds
\end{enumerate}

\vspace{0.3cm}
\muted{$\Rightarrow$ Chinese mutual fund social media livestreaming provides all three.}

\end{frame}

\section{Setting and Data}
% ===========================================================================

\transitionslide{Setting and Data}

\begin{frame}{Setting: Mutual Fund Livestreaming in China}

\textbf{A novel, large-scale marketing channel:}
\begin{itemize}
  \item Beginning 2020 (COVID-19 disrupted traditional sales channels), fund families adopted \hl{real-time broadcasts} on retail platforms
  \item Tiantian Fund — leading third-party distribution platform — hosts the broadcasts
  \item AMAC formalized regulatory guidance in 2021 (disclosure, compliance controls)
\end{itemize}

\vspace{0.35cm}
\textbf{Key institutional features:}
\begin{itemize}
  \item \textbf{Family-level production:} each family runs its own channel with dedicated staff
  \item \textbf{Sharp selection:} families feature $<4$ funds per broadcast from $25+$ products
  \item \textbf{Shopping cart:} viewers can purchase featured funds in one click during the broadcast
\end{itemize}

\vspace{0.35cm}
\begin{keybox}
  \textbf{Identification gift:} within the same family, some funds are featured while others are not — in the \emph{same quarter}.
\end{keybox}

\end{frame}

% ---------------------------------------------------------------------------

\begin{frame}{Data and Sample}

\begin{columns}[T]
\begin{column}{0.52\textwidth}
  \textbf{Livestream data (Tiantian Fund):}
  \begin{itemize}
    \item \hl{25,685} livestreaming events
    \item July 2020 -- June 2025
    \item Treatment: fund appears in shopping cart $\Rightarrow$ ``featured''
  \end{itemize}
  \vspace{0.3cm}
  \textbf{Fund universe:}
  \begin{itemize}
    \item \hl{5,809} actively managed equity/hybrid funds
    \item \hl{180} fund families
    \item \hl{90,850} fund-quarter observations
    \item Exclude: index, closed-end, QDII, umbrella funds
  \end{itemize}
\end{column}
\begin{column}{0.46\textwidth}
  \textbf{Key variables:}
  \vspace{0.15cm}

  \textit{Fund flow} (return-adjusted):
  \[
  \text{Flow}_{i,q} = \frac{\text{TNA}_{i,q} - \text{TNA}_{i,q-1}(1+r_{i,q})}{\text{TNA}_{i,q-1}}
  \]

  \textit{Family flow}: AUM-weighted aggregate across all family funds.

  \vspace{0.2cm}
  \textit{Descriptives:}
  \begin{itemize}
    \item Mean flow: 0.38\%, SD: 42\%
    \item \hl{14\%} of fund-quarters are featured
    \item Featured funds outperform siblings pre-event
  \end{itemize}
\end{column}
\end{columns}

\end{frame}

% ---------------------------------------------------------------------------

\begin{frame}{Empirical Design}

\textbf{Two estimating equations:}

\vspace{0.2cm}
\textit{Family-level TWFE} (does the family grow?):
\[
\text{FamilyFlow}_{j,q} = \alpha_j + \delta_q + \beta_1\,\text{Livestream}_{j,q} + \mathbf{X}'_{j,q-1}\Gamma + \varepsilon_{j,q}
\]

\vspace{0.2cm}
\textit{Fund-level TWFE} (who gains, who loses?):
\[
\text{FundFlow}_{i,q} = \alpha_i + \delta_q + \beta_1\,\underbrace{\text{FundLivestream}_{i,q}}_{\text{fund featured}} + \beta_2\,\underbrace{\text{FamilyLivestream(Oth)}_{i,q}}_{\text{sibling featured}} + \mathbf{X}'_{i,q-1}\Gamma + \varepsilon_{i,q}
\]

\vspace{0.3cm}
\textbf{Key design features:}
\begin{itemize}
  \item Fund and family fixed effects absorb time-invariant unobservables
  \item Year-quarter fixed effects control for aggregate market conditions
  \item Standard errors clustered at the family level
  \item Robustness: family-quarter FE (absorbs all time-varying family shocks)
\end{itemize}

\end{frame}

\begin{frame}{Robustness: Family-Quarter Fixed Effects}

\textbf{Strategy:} Replace fund + quarter FE with \hl{family $\times$ quarter FE}
\begin{itemize}
  \item Absorbs \emph{all} time-varying family characteristics (family flow, family performance, market conditions)
  \item Mechanically forces AUM-weighted within-family flows to sum to zero (since Table~1 establishes family null)
\end{itemize}

\vspace{0.35cm}
\begin{center}
\begin{tabular}{lccc}
  \toprule
   & (1) & (2) & (3) \\
   & First & Post & Any \\
  \midrule
  Fund Livestream & 15.40*** & 16.17*** & 17.59*** \\
   & (2.12) & (2.08) & (2.25) \\
  \bottomrule
\end{tabular}
\end{center}

\vspace{0.25cm}
\begin{resultbox}
Coefficients \hl{larger} than TWFE by $\approx 3$pp: the family-quarter FE forces featured-fund gains to equal non-featured-fund losses. The \textbf{17.6pp} is the \emph{full reallocation magnitude}.
\end{resultbox}

\begin{frame}{Interpretation: Fund Families as Open Market Operators}

\textbf{Pattern before the livestream:}
\begin{itemize}
  \item Featured funds significantly \hl{outperform} non-featured siblings in the months leading up to the event
  \item Non-featured funds underperform $\Rightarrow$ they are \emph{overcapitalized}
\end{itemize}

\textbf{Pattern after the livestream:}
\begin{itemize}
  \item Capital flows into featured funds erode their outperformance
  \item Capital flows out of non-featured funds improve their performance
  \item Performance gap compresses — consistent with \citet{berk2004mutual}
\end{itemize}

\vspace{0.3cm}
\begin{keybox}
\textbf{Mechanism:} Families have superior information about which funds are undercapitalized. Livestreaming steers existing investors from overcapitalized to undercapitalized funds — functioning as an \hl{internal open market operation} \citep{berk2017}.

\medskip
\textbf{Benefit:} Retains investors who might otherwise search outside the family.
\end{keybox}

\end{frame}

\section{Communication Strategies}
% ===========================================================================

\transitionslide{Results III: Which Livestreams Attract Outside Capital?}

\begin{frame}{What Distinguishes High- vs.\ Low-Flow Livestreamers?}

\textbf{Question:} While most livestreaming is intra-family reallocation, some families do attract \emph{new} outside capital. What do they do differently?

\vspace{0.3cm}
\textbf{Approach:}
\begin{itemize}
  \item Define abnormal family flows as residuals from a regression of family flows on livestream intensity + controls
  \item Sort into High (top 5\%) vs.\ Low (bottom 5\%) within each quarterly cross-section
  \item Randomly sample 5 transcripts per family $\Rightarrow$ \hl{795 transcripts} for content analysis
\end{itemize}

\vspace{0.3cm}
\textbf{LLM scoring (Qwen-2.5-32B):}
\begin{itemize}
  \item 16 dimensions: sentiment, emotional intensity, sales pressure, past performance, social proof, educational content, risk transparency, community building, \ldots
  \item Each scored 1--10 using a fixed rubric (segment-level, then averaged)
  \item Rubric fixed ex ante $\Rightarrow$ measurement error orthogonal to strategies under study
\end{itemize}

\end{frame}

\section{Competitive Equilibrium}
% ===========================================================================

\begin{frame}{Competitive Pressure and the Defensive Equilibrium}

\textbf{If livestreaming generates no family-level gains, why continue?}

\vspace{0.3cm}
\textbf{Market-level competitive pressure:}
\begin{itemize}
  \item When rival families increase livestreaming intensity, non-livestreaming families experience \hl{relative outflows}
  \item A one-SD increase in market livestreaming predicts $\approx 27$pp flow losses for families that do not match competitors' efforts
\end{itemize}

\vspace{0.35cm}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\textbf{Within families:}
\begin{itemize}
  \item Own-fund livestreaming: $+7.2$pp flows
  \item Sibling livestreaming: $-4.9$pp flows
  \item Net family effect: $\approx 0$
\end{itemize}
\end{column}
\begin{column}{0.48\textwidth}
\textbf{Across families:}
\begin{itemize}
  \item Failing to livestream when competitors do $\Rightarrow$ market share loss
\end{itemize}
\end{column}
\end{columns}

\vspace{0.4cm}
\begin{keybox}
\textbf{Prisoner's dilemma:} families must livestream \emph{not to attract new investors} but to defend existing assets from rivals. Industry equilibrium: too much marketing \citep{roussanov2021marketing}.
\end{keybox}

\end{frame}

\section{Conclusion}
% ===========================================================================

\transitionslide{Conclusion}

\begin{frame}{Summary of Findings}

\begin{enumerate}
  \setlength\itemsep{0.4cm}

  \item[\hl{(1)}] \textbf{Zero-sum at the family level.} Livestreaming generates large fund-level inflows (+14pp) but \emph{zero} family-level flow or revenue change. Featured funds gain at the expense of non-featured siblings.

  \item[\hl{(2)}] \textbf{Intra-family optimization.} Families promote recently outperforming (undercapitalized) funds to redirect capital from overcapitalized siblings. Post-event performance converges — consistent with Berk-Green rational learning.

  \item[\hl{(3)}] \textbf{Defensive arms race.} Families livestream to defend market share, not to expand it. When rivals promote more, non-promoters lose flows. Industry is trapped in wasteful promotional competition.

  \item[\hl{(4)}] \textbf{Hard information wins.} The rare cases of successful inter-family capital attraction rely exclusively on past performance and social proof — not sentiment or engagement.
\end{enumerate}

\end{frame}

% ---------------------------------------------------------------------------

\begin{frame}{Broader Implications}

\textbf{For the mutual fund literature:}
\begin{itemize}
  \item Challenges the implicit assumption that fund-level marketing = family growth
  \item Flow-performance sensitivities and marketing effectiveness must be evaluated at the family level
\end{itemize}

\vspace{0.3cm}
\textbf{For investor welfare:}
\begin{itemize}
  \item Livestreaming facilitates learning and improves capital allocation \emph{within} families
  \item But aggregate marketing may be socially excessive: competitive arms race with no net new capital formation
  \item Welfare implications of this technology remain an open question
\end{itemize}

\vspace{0.3cm}
\textbf{For financial regulation:}
\begin{itemize}
  \item AMAC's 2021 guidance correctly targets investor education and compliance
  \item Our evidence suggests livestreaming can be informative (hard info channel) but is also used defensively
\end{itemize}

\end{frame}
