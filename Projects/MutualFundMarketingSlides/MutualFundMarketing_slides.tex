\section{Motivation}
% =============================================================================

\begin{frame}{Mutual Fund Marketing: What We Know}

\begin{wideitemize}
  \item Asset management has grown dramatically — AUM and fund counts have surged
  over three decades; \hl{investor search costs} are a first-order friction
  (Hortaçsu \& Syverson, 2004; Huang, Wei \& Yan, 2007)

  \item U.S.\ mutual funds spend roughly \hl{one-third of fee revenue on marketing};
  advertising, broker distribution, and visibility drive significant fund-level inflows
  (Sirri \& Tufano, 1998; Gallaher, Kaniel \& Starks, 2008)

  \item A recent literature studies social media marketing of funds:
  De Haan et al.\ (2025) show that featured funds on Chinese platforms receive
  \hl{significant capital inflows}
\end{wideitemize}

\vspace{0.4cm}
\begin{block}{Implicit assumption in the literature}
  Fund-level marketing $\;\Rightarrow\;$ family grows. \quad Is this right?
\end{block}

\end{frame}

% -----------------------------------------------------------------------

\begin{frame}{The Missing Family-Level Test}

\begin{wideitemize}
  \item Marketing \textbf{expands the family} only if inflows represent \hl{new money}
  from outside the family

  \item If marketing primarily \hl{reallocates existing assets} within the family,
  then fund-level effects do not translate into family growth

  \item Prior work evaluates marketing at the fund level despite
  \textbf{family-level decision-making} (Gallaher et al., 2008 is the rare exception)
\end{wideitemize}

\bigskip
\textbf{What we need:}
\begin{enumerate}
  \item A marketing channel with \hl{sharp within-family variation in exposure}
  \item Joint observation of fund-level \emph{and} family-level flows
  \item Transcript-level data to understand what content drives outside capital
\end{enumerate}

\bigskip
\muted{$\Rightarrow$ Chinese mutual fund social media livestreaming provides all three.}

\end{frame}

\begin{frame}{Institutional Setting: Mutual Fund Livestreaming}

\begin{columns}[T]
\begin{column}{0.55\textwidth}
\textbf{A large-scale novel marketing channel:}
\begin{itemize}
  \item From 2020, COVID-19 disrupted traditional sales; fund families adopted
  \hl{real-time broadcasts} on retail platforms
  \item \textbf{Tiantian Fund} — leading third-party platform — hosts the broadcasts
  \item AMAC formalized regulatory guidance in 2021
\end{itemize}

\vspace{0.25cm}
\textbf{Key institutional features:}
\begin{itemize}
  \item Production is \hl{family-level}: dedicated staff, family channel
  \item Sharp selection: $<4$ funds featured per broadcast from $25+$ products
  \item \hl{Shopping cart}: viewers purchase featured funds in one click
  \item Compliance approval required in advance — timing is planned
\end{itemize}
\end{column}
\begin{column}{0.43\textwidth}
\begin{alertblock}{Identification gift}
  Within the same family and same quarter, some funds are featured while others are not.

  \bigskip
  This within-family, within-quarter variation identifies both the direct effect on featured funds \emph{and} the spillover to non-featured siblings.
\end{alertblock}
\end{column}
\end{columns}

\end{frame}

% -----------------------------------------------------------------------

\begin{frame}{Data and Sample}

\begin{columns}[T]
\begin{column}{0.50\textwidth}
  \textbf{Livestream data (Tiantian Fund):}
  \begin{itemize}
    \item \hl{25,685} livestreaming events, July 2020 -- June 2025
    \item Treatment: fund in shopping cart $=$ ``featured''
  \end{itemize}

  \vspace{0.3cm}
  \textbf{Fund universe:}
  \begin{itemize}
    \item \hl{5,809} actively managed equity/hybrid funds
    \item \hl{180} fund families
    \item \hl{90,850} fund-quarter observations
    \item Exclude: index, closed-end, QDII, umbrella funds
  \end{itemize}

  \vspace{0.3cm}
  \textbf{Descriptives:}
  \begin{itemize}
    \item Mean flow: 0.38\%, SD: 42\%
    \item \hl{14\%} of fund-quarters are featured
    \item Featured funds outperform siblings pre-event
  \end{itemize}
\end{column}
\begin{column}{0.48\textwidth}
  \textbf{Flow measurement:}

  \vspace{0.1cm}
  \textit{Fund flow} (return-adjusted):
  \[
  \text{Flow}_{i,q} = \frac{\text{TNA}_{i,q} - \text{TNA}_{i,q-1}(1+r_{i,q})}{\text{TNA}_{i,q-1}}
  \]

  \vspace{0.1cm}
  \textit{Family flow}: AUM-weighted aggregate of fund flows within family $f$:
  \[
  \text{FamilyFlow}_{f,q} = \frac{\sum_{i \in f}\!\left[\text{TNA}_{i,q} - \text{TNA}_{i,q-1}(1\!+\!r_{i,q})\right]}{\sum_{i \in f} \text{TNA}_{i,q-1}}
  \]

  \vspace{0.1cm}
  \muted{\small Captures whether livestreaming expands the family's investor base or merely redistributes assets.}
\end{column}
\end{columns}

\end{frame}

% -----------------------------------------------------------------------

\begin{frame}{Empirical Design}

\textbf{Two estimating equations:}

\vspace{0.2cm}
\textit{Family-level TWFE} --- does the family grow?
\[
\text{FamilyFlow}_{f,q} = \alpha_f + \delta_q
  + \beta_1\,\text{Livestream}_{f,q}
  + \mathbf{X}'_{f,q-1}\Gamma + \varepsilon_{f,q}
\]

\vspace{0.15cm}
\textit{Fund-level TWFE} --- who gains, who loses?
\[
\text{FundFlow}_{i,q} = \alpha_i + \delta_q
  + \beta_1\underbrace{\text{FundLivestream}_{i,q}}_{\text{fund $i$ is featured}}
  + \beta_2\underbrace{\text{FamilyLivestream(Oth)}_{i,q}}_{\text{sibling fund is featured}}
  + \mathbf{X}'_{i,q-1}\Gamma + \varepsilon_{i,q}
\]

\vspace{0.3cm}
\begin{columns}[T]
\begin{column}{0.47\textwidth}
\textbf{Fixed effects:}
\begin{itemize}
  \item Fund / family FE absorb time-invariant unobservables
  \item Year-quarter FE control for aggregate conditions
\end{itemize}
\end{column}
\begin{column}{0.47\textwidth}
\textbf{Robustness:}
\begin{itemize}
  \item Family $\times$ quarter FE: absorbs all time-varying family shocks
  \item SE clustered at family level throughout
\end{itemize}
\end{column}
\end{columns}

\end{frame}

\begin{frame}{Robustness: Family $\times$ Quarter Fixed Effects}

\textbf{Replace fund + quarter FE with family $\times$ quarter FE:}
\begin{itemize}
  \item Absorbs \emph{all} time-varying family characteristics: aggregate family flow, family performance, market conditions
  \item Mechanically forces AUM-weighted within-family flows to sum to the family's aggregate flow
  \item Since family aggregate = 0 (Table 1), the coefficient on featured funds = \hl{full reallocation magnitude}
\end{itemize}

\vspace{0.3cm}
\begin{center}
\begin{tabular}{lcc}
  \toprule

   & (1.85) & (2.25) \\
  \bottomrule

\end{center}

\vspace{0.2cm}
\begin{block}{Interpretation}
  The \hl{+17.6 pp} under family$\times$quarter FE represents the dollar-for-dollar reallocation: every dollar flowing into featured funds is offset by an equal dollar flowing out of non-featured siblings.
\end{block}

\begin{frame}{Interpretation: Families as Open Market Operators}

\textbf{Pattern before the livestream:}
\begin{itemize}
  \item Featured funds significantly \hl{outperform} siblings in months preceding the event
  \item Non-featured funds underperform $\;\Rightarrow\;$ they are \textit{overcapitalized} (Berk \& Green, 2004)
\end{itemize}

\vspace{0.2cm}
\textbf{Pattern after the livestream:}
\begin{itemize}
  \item Inflows into featured funds erode their outperformance
  \item Outflows from non-featured funds improve their performance
  \item Performance gap \hl{compresses} --- consistent with flow-performance equilibrium
\end{itemize}

\vspace{0.25cm}
\begin{block}{Mechanism: Internal open market operation (Berk, 2017)}
  Families have superior information about which funds are undercapitalized.
  Livestreaming \hl{steers existing investors} from overcapitalized to undercapitalized funds ---
  without attracting new outside capital.

  \smallskip
  \textbf{Benefit:} Retains investors who might otherwise search outside the family.
\end{block}

\end{frame}

\begin{frame}{Which Livestreams Actually Attract Outside Capital?}

\textbf{Question:} While most livestreaming is intra-family reallocation, some families do attract new outside money. What distinguishes them?

\bigskip
\textbf{Approach:}
\begin{itemize}
  \item Define \textit{abnormal family flows} = residuals from regression of family flows on livestream intensity + controls
  \item Sort into High (top 5\%) vs.\ Low (bottom 5\%) within each quarterly cross-section
  \item Randomly sample 5 transcripts per family $\;\Rightarrow\;$ \hl{795 transcripts}
\end{itemize}

\bigskip
\textbf{LLM scoring (Qwen-2.5-32B):}
\begin{itemize}
  \item Score each transcript on \hl{16 dimensions}: sentiment, emotional intensity, sales pressure, past performance, social proof, educational content, risk transparency, community building, \ldots
  \item Each scored 1--10 using a \textit{fixed rubric defined ex ante}
  \item Segment-level (10k chars), averaged to transcript level
\end{itemize}

\end{frame}

\section{Competitive Equilibrium}
% =============================================================================

\begin{frame}{Competitive Pressure: Why Families Must Livestream}

\textbf{If own-livestreaming generates no family-level gains, why do families persist?}

\bigskip
\begin{columns}[T]
\begin{column}{0.47\textwidth}
\textbf{Within families (own vs.\ sibling):}
\begin{itemize}
  \item Own-fund livestreaming: \good{$+$7.2 pp} flows
  \item Sibling livestreaming: \bad{$-$4.9 pp} flows
  \item Net family effect: $\approx 0$
\end{itemize}

\vspace{0.3cm}
\textbf{Across families (market competition):}
\begin{itemize}
  \item When rivals increase livestreaming, non-livestreamers lose \hl{$\approx 27$ pp} flows per 1-SD increase in market activity
  \item Own market share predicts $+$620 pp flows per unit increase
\end{itemize}
\end{column}
\begin{column}{0.50\textwidth}
\begin{block}{Prisoner's dilemma}
  Families must livestream \textbf{not to attract new investors} but to \hl{defend existing assets} from rivals who are more actively marketing.

  \bigskip
  Unilateral exit from livestreaming results in immediate flow losses.

  \bigskip
  Industry is trapped in a wasteful promotional arms race (Roussanov, Russ \& Wei, 2021).
\end{block}
\end{column}
\end{columns}

\end{frame}

\section{Conclusion}
% =============================================================================

\begin{transitionframe}
  \begin{center}
    {\Large \textbf{\textcolor{white}{Conclusion}}}
  \end{center}
\end{transitionframe}

\begin{frame}{Summary}

\begin{enumerate}\addtolength{\itemsep}{0.3cm}

  \item[\hlg{(1)}] \textbf{Zero-sum at the family level.}
    Livestreaming generates large fund-level inflows (\hl{+14 pp}) but \textbf{zero} family-level flow or revenue change. Featured funds gain at the direct expense of non-featured siblings.

  \item[\hlg{(2)}] \textbf{Intra-family capital optimization.}
    Families promote recently outperforming (undercapitalized) funds. Post-event performance converges --- consistent with Berk-Green rational learning. Livestreaming is an \hl{internal open market operation}.

  \item[\hlg{(3)}] \textbf{Defensive arms race.}
    Families livestream to \hl{defend market share}, not to expand it. Market-wide competition creates a prisoner's dilemma: exit means flow losses, stay means zero net gain.

  \item[\hlg{(4)}] \textbf{Hard information wins.}
    The rare cases of successful inter-family capital attraction rely exclusively on \hl{past performance and social proof} --- not sentiment or engagement.

\end{enumerate}

\end{frame}

% -----------------------------------------------------------------------

\begin{frame}{Broader Implications}

\begin{wideitemize}

  \item \textbf{For the mutual fund literature:} Fund-level marketing effects
  $\;\not\Rightarrow\;$ family growth. Flow-performance sensitivities must be evaluated at the family level.

  \item \textbf{For investor welfare:} Livestreaming facilitates learning and
  improves within-family capital allocation --- but aggregate marketing may be
  socially excessive given the competitive arms race with no net new capital formation.

  \item \textbf{For policy:} AMAC's 2021 guidance correctly targets investor
  education and compliance. Our evidence confirms livestreaming can be informative
  (hard-info channel) but also has a defensive, rent-defending motive.

\end{wideitemize}
